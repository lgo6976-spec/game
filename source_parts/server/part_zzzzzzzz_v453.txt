def mafia_role_plan(n):
    mcount=1 if n<=6 else 2 if n<=10 else 3
    roles=['마피아']*mcount+['경찰']
    if n>=6:roles.append('의사')
    roles+=['시민']*(n-len(roles))
    return roles

def mafia_role_summary_text(roles):
    c=Counter(roles);parts=[]
    for r in ('마피아','경찰','의사','시민'):
        if c.get(r):parts.append(f'{r} {c[r]}')
    return ' · '.join(parts)

def setup_mafia(room):
    pids=shuffled_active(room);roles=mafia_role_plan(len(pids));random.shuffle(roles)
    room['assignments']={};room['pending']={};room['submissions']={};room['votes']={};room['round_result']=None;room['round_no']=1
    for p,r in zip(pids,roles):
        room['players'][p]['alive']=True
        mission='밤에는 제거 대상을 고르세요.' if r=='마피아' else '밤에는 한 명을 조사하세요.' if r=='경찰' else '밤에는 한 명을 보호하세요.' if r=='의사' else '낮 토론에서 마피아를 찾아내세요.'
        make_assignment(room,p,r,mission,None,{'mafia_notice':''})
    maf=[p for p in pids if room['assignments'][p]['role']=='마피아']
    for p in maf:
        mates=[room['players'][x]['name'] for x in maf if x!=p]
        append_private_notice(room,p,'마피아 팀: '+(', '.join(mates) if mates else '당신 혼자입니다.'))
    summary=mafia_role_summary_text(roles)
    room['game_data']={
        'start_scores':{p:room['players'][p]['score'] for p in pids},'mafia_day':1,'mafia_stage':'ready','mafia_actions':{},
        'doctor_last_target':None,'mafia_role_summary':summary,'mafia_role_counts':dict(Counter(roles)),
        'mafia_candidates':[],'mafia_nomination_votes':{},'mafia_final_votes':{},'mafia_no_elim_streak':0,
        'mafia_hard_end_at':now()+22*60
    }
    room['public']={'event_title':'마피아 준비','event_text':'역할 구성 · '+summary,'story':None,'special':None}
    room['phase']='countdown';room['phase_started']=now();room['deadline']=now()+5;room['interaction_deadline']=None;room['paused']=False;room['pause_remaining']=None;room['paused_at']=None

def mafia_discussion_seconds(room):
    n=len(mafia_alive(room));base=55 if n<=5 else 65 if n==6 else 80 if n<=8 else 95 if n<=10 else 110
    day=max(1,int(room.get('game_data',{}).get('mafia_day',1)))
    return max(45,base-(day-1)*8)

def resolve_mafia_night(room):
    gd=room['game_data'];actions=gd.get('mafia_actions',{});alive=mafia_alive(room);maf=[p for p in alive if room['assignments'][p]['role']=='마피아'];kills=[actions[p] for p in maf if p in actions]
    kill=None
    if kills:
        c=Counter(kills);mx=max(c.values());kill=random.choice([p for p,v in c.items() if v==mx])
    doctor=next((p for p in alive if room['assignments'][p]['role']=='의사'),None);saved=actions.get(doctor) if doctor else None
    if doctor and saved:gd['doctor_last_target']=saved
    police=next((p for p in alive if room['assignments'][p]['role']=='경찰'),None)
    if police and police in actions:
        t=actions[police];is_maf=room['assignments'][t]['role']=='마피아';append_private_notice(room,police,f'{gd["mafia_day"]}일차 조사 · {room["players"][t]["name"]}: '+('마피아입니다.' if is_maf else '마피아가 아닙니다.'))
    if kill and kill!=saved:
        room['players'][kill]['alive']=False;role=room['assignments'][kill]['role'];text=f'{room["players"][kill]["name"]}님이 밤에 제거되었습니다. 역할은 {role}였습니다.'
    else:text='지난밤 아무도 제거되지 않았습니다.'
    room['round_result']={'stage':'mafia_morning','title':'아침','text':text};room['public']['event_title']='☀️ 아침';room['public']['event_text']=text
    if mafia_check_win(room):return
    gd['mafia_stage']='morning';room['interaction_deadline']=now()+7;room['deadline']=room['interaction_deadline']

def start_mafia_discuss(room):

def start_mafia_nomination(room):
    gd=room['game_data'];gd['mafia_stage']='nominate';gd['mafia_nomination_votes']={};room['votes']={};room['round_result']=None

    room['interaction_deadline']=now()+20;room['deadline']=room['interaction_deadline']

def mafia_nominate(room,pid,target):
    gd=room['game_data'];alive=mafia_alive(room)
    if gd.get('mafia_stage')!='nominate' or pid not in alive:raise ValueError('지금은 용의자 지목 단계가 아닙니다.')
    votes=gd.setdefault('mafia_nomination_votes',{})
    if pid in votes:raise ValueError('이미 지목했습니다.')
    if target not in alive or target==pid:raise ValueError('다른 생존자를 선택하세요.')
    votes[pid]=target
    if len(votes)>=len(alive):resolve_mafia_nomination(room)

def resolve_mafia_nomination(room):
    gd=room['game_data'];alive=mafia_alive(room);votes={p:t for p,t in gd.get('mafia_nomination_votes',{}).items() if p in alive and t in alive and p!=t}
    if not votes:
        candidates=random.sample(alive,min(2,len(alive)))
    else:
        cnt=Counter(votes.values());ordered=cnt.most_common();first=ordered[0][1];top=[p for p,c in ordered if c==first]
        if len(top)>=2:candidates=random.sample(top,min(2,len(top)))
        else:
            candidates=[top[0]];second=[p for p,c in ordered if c<first]
            if second:
                second_count=max(cnt[p] for p in second);pool=[p for p in second if cnt[p]==second_count];candidates.append(random.choice(pool))
            elif len(alive)>1:candidates.append(random.choice([p for p in alive if p!=top[0]]))
    gd['mafia_candidates']=candidates;gd['mafia_stage']='defense';names=' vs '.join(room['players'][p]['name'] for p in candidates)
    counts=Counter(votes.values());room['round_result']={'stage':'mafia_nomination_result','title':'결선 후보','text':names,'counts':[{'name':room['players'][p]['name'],'count':counts[p]} for p in candidates]}
    room['public']['event_title']='최후변론';room['public']['event_text']=names+' · 후보들은 차례로 짧게 변론하세요.';room['interaction_deadline']=now()+(14 if len(candidates)==1 else 24);room['deadline']=room['interaction_deadline']

def start_mafia_final_vote(room):
    gd=room['game_data'];gd['mafia_stage']='final_vote';gd['mafia_final_votes']={};room['round_result']=None;forced=gd.get('mafia_no_elim_streak',0)>=1
    room['public']['event_title']='결선투표';room['public']['event_text']='결선 후보 중 탈락시킬 사람을 고르세요.'+(' 이번에는 반드시 한 명이 탈락합니다.' if forced else ' 확신이 없으면 살려둘 수 있습니다.')
    room['interaction_deadline']=now()+20;room['deadline']=room['interaction_deadline'];gd['mafia_abstain_allowed']=not forced

def mafia_final_vote(room,pid,target):
    gd=room['game_data'];alive=mafia_alive(room);candidates=gd.get('mafia_candidates',[]);votes=gd.setdefault('mafia_final_votes',{})
    if gd.get('mafia_stage')!='final_vote' or pid not in alive:raise ValueError('지금은 결선투표 단계가 아닙니다.')
    if pid in votes:raise ValueError('이미 결선투표했습니다.')
    valid=set(candidates)
    if gd.get('mafia_abstain_allowed'):valid.add('abstain')
    if target not in valid:raise ValueError('결선 후보 중 한 명을 선택하세요.')
    if target==pid:raise ValueError('자기 자신에게는 투표할 수 없습니다.')
    votes[pid]=target
    if len(votes)>=len(alive):resolve_mafia_final_vote(room)

def resolve_mafia_final_vote(room):
    gd=room['game_data'];alive=mafia_alive(room);candidates=gd.get('mafia_candidates',[]);votes={p:t for p,t in gd.get('mafia_final_votes',{}).items() if p in alive and (t in candidates or t=='abstain')};cnt=Counter(votes.values());elim=None
    candidate_counts={p:cnt[p] for p in candidates};best=max(candidate_counts.values() or [0]);tops=[p for p,v in candidate_counts.items() if v==best and best>0];abstain=cnt.get('abstain',0)
    if len(tops)==1 and (not gd.get('mafia_abstain_allowed') or best>abstain):elim=tops[0]
    if elim:
        room['players'][elim]['alive']=False;role=room['assignments'][elim]['role'];text=f'{room["players"][elim]["name"]}님이 탈락했습니다. 역할은 {role}였습니다.';gd['mafia_no_elim_streak']=0
    else:
        text='결선 결과 한 명을 확정하지 못해 아무도 탈락하지 않았습니다.';gd['mafia_no_elim_streak']=gd.get('mafia_no_elim_streak',0)+1
    rows=[{'name':room['players'][p]['name'],'count':cnt[p]} for p in candidates]
    if gd.get('mafia_abstain_allowed'):rows.append({'name':'살려둔다','count':abstain})
    room['round_result']={'stage':'mafia_vote_result','title':'결선 결과','text':text,'counts':rows};room['public']['event_title']='투표 결과';room['public']['event_text']=text
    if mafia_check_win(room):return
    gd['mafia_stage']='vote_result';room['interaction_deadline']=now()+7;room['deadline']=room['interaction_deadline']

def mafia_night_action(room,pid,target):
    gd=room['game_data'];alive=mafia_alive(room)
    if gd.get('mafia_stage')!='night' or pid not in alive:raise ValueError('지금은 밤 행동을 할 수 없습니다.')
    role=room['assignments'][pid]['role']
    if role not in ('마피아','경찰','의사'):raise ValueError('현재 역할에는 밤 행동이 없습니다.')
    if pid in gd.setdefault('mafia_actions',{}):raise ValueError('이미 밤 행동을 선택했습니다.')
    if target not in alive:raise ValueError('살아있는 참가자를 선택하세요.')
    if role in ('마피아','경찰') and target==pid:raise ValueError('자기 자신은 선택할 수 없습니다.')
    if role=='마피아' and room['assignments'][target]['role']=='마피아':raise ValueError('같은 마피아는 제거 대상으로 선택할 수 없습니다.')
    if role=='의사' and target==gd.get('doctor_last_target'):raise ValueError('의사는 같은 사람을 연속으로 보호할 수 없습니다.')
    gd['mafia_actions'][pid]=target
    required=[p for p in alive if room['assignments'][p]['role'] in ('마피아','경찰','의사')]
    if all(p in gd['mafia_actions'] for p in required):resolve_mafia_night(room)

# compatibility: old mafia_vote action becomes final vote when called by stale clients.
def mafia_vote(room,pid,target):
    gd=room.get('game_data',{})
    if gd.get('mafia_stage')=='nominate':return mafia_nominate(room,pid,target)
    if gd.get('mafia_stage')=='final_vote':return mafia_final_vote(room,pid,target)
    raise ValueError('지금은 투표 단계가 아닙니다.')

_V453_DO_ACTION=do_action
def do_action(room,pid,data):
    t=data.get('type')
    if t=='mafia_nominate':
        if room.get('phase')!='play' or room.get('game_id')!='mafia' or room.get('paused'):raise ValueError('지금은 지목할 수 없습니다.')
        return mafia_nominate(room,pid,data.get('target'))
    if t=='mafia_final_vote':
        if room.get('phase')!='play' or room.get('game_id')!='mafia' or room.get('paused'):raise ValueError('지금은 결선투표할 수 없습니다.')
        return mafia_final_vote(room,pid,data.get('target'))
    return _V453_DO_ACTION(room,pid,data)

# Mafia owns its staged timing, replacing the v4.5.1 stage dispatcher.
_V453_TICK=tick
def tick(room):
    if room.get('game_id')!='mafia':return _V453_TICK(room)
    if room.get('phase')=='countdown':return _V453_TICK(room)
    if room.get('phase')!='play' or room.get('paused'):return
    gd=room.get('game_data',{});ts=now();st=gd.get('mafia_stage')
    if ts>=gd.get('mafia_hard_end_at',1e18):
        # Avoid an endless social stalemate: force the next daytime runoff to eliminate someone.
        gd['mafia_no_elim_streak']=1
    if st=='night' and ts>=room.get('interaction_deadline',1e18):resolve_mafia_night(room)
    elif st=='morning' and ts>=room.get('interaction_deadline',1e18):start_mafia_discuss(room)
    elif st=='discuss':
        rem=room.get('interaction_deadline',ts)-ts
        if rem<=20 and not gd.get('mafia_discuss_warning'):
            gd['mafia_discuss_warning']=True;room['public']['event_title']='토론 20초 남음';room['public']['event_text']='새 주장보다 지금까지 나온 근거를 정리하고 용의자를 좁히세요.'
        if ts>=room.get('interaction_deadline',1e18):start_mafia_nomination(room)
    elif st=='nominate' and ts>=room.get('interaction_deadline',1e18):resolve_mafia_nomination(room)
    elif st=='defense' and ts>=room.get('interaction_deadline',1e18):start_mafia_final_vote(room)
    elif st=='final_vote' and ts>=room.get('interaction_deadline',1e18):resolve_mafia_final_vote(room)
    elif st=='vote_result' and ts>=room.get('interaction_deadline',1e18):gd['mafia_day']+=1;start_mafia_night(room)
