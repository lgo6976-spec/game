def setup_reaction(room):
    pids=active_pids(room);room['assignments']={};room['pending']={};room['submissions']={};room['votes']={};room['round_result']=None;room['round_no']=1
    for p in pids:make_assignment(room,p,'플레이어','신호가 뜰 때까지 기다리세요.',None,{'reaction_status':'ready'})
    room['game_data']={'start_scores':{p:room['players'][p]['score'] for p in pids},'reaction_index':0,'reaction_total':8,'reaction_stage':'ready','reaction_last_mode':None}
    room['public']={'event_title':'반응속도 준비','event_text':'성급하게 누르면 그 라운드는 점수를 받을 수 없습니다.','story':None,'special':None}
    room['phase']='countdown';room['phase_started']=now();room['deadline']=now()+4;room['interaction_deadline']=None;room['paused']=False;room['pause_remaining']=None;room['paused_at']=None

def start_reaction_round(room):
    gd=room['game_data'];idx=gd['reaction_index'];pids=active_pids(room);room['submissions']={};room['round_result']=None;room['round_no']=idx+1
    mode='stop' if gd.get('reaction_last_mode')!='stop' and random.random()<.28 else 'go';gd['reaction_last_mode']=mode;gd['reaction_mode']=mode;gd['reaction_stage']='wait';gd['reaction_false']={};gd['reaction_cue_at']=now()+random.uniform(1.6,3.8);gd['reaction_resolve_at']=gd['reaction_cue_at']+1.7
    for p in pids:make_assignment(room,p,'플레이어','기다리세요…',None,{'reaction_status':'wait'})
    room['public']['event_title']=f'ROUND {idx+1}/{gd["reaction_total"]}';room['public']['event_text']='아직 누르지 마세요.';room['deadline']=gd['reaction_resolve_at'];room['interaction_deadline']=gd['reaction_resolve_at']

def cue_reaction(room):
    gd=room['game_data'];mode=gd['reaction_mode'];gd['reaction_stage']='cue';room['public']['event_title']='GO!' if mode=='go' else 'STOP!';room['public']['event_text']='지금 누르세요!' if mode=='go' else '누르면 안 됩니다!'
    for p in active_pids(room):
        a=room['assignments'].get(p)
        if a:a['mission']='지금 누르세요!' if mode=='go' else '절대 누르지 마세요!';a.setdefault('extra',{})['reaction_status']=mode

def reaction_tap(room,pid):
    gd=room['game_data'];stage=gd.get('reaction_stage')
    if pid in room.get('submissions',{}) or pid in gd.setdefault('reaction_false',{}):raise ValueError('이번 라운드에는 이미 반응했습니다.')
    if stage=='wait':
        gd['reaction_false'][pid]=now();append_private_notice(room,pid,'너무 빨랐습니다. 이번 라운드는 0점입니다.');return
    if stage!='cue':raise ValueError('지금은 누를 수 없습니다.')
    room['submissions'][pid]={'ts':now(),'ms':max(0,int((now()-gd['reaction_cue_at'])*1000))}
    if gd['reaction_mode']=='go' and len(room['submissions'])+len(gd.get('reaction_false',{}))>=len(active_pids(room)):resolve_reaction(room)

def resolve_reaction(room):
    gd=room['game_data'];mode=gd['reaction_mode'];pids=active_pids(room);false=set(gd.get('reaction_false',{}));subs=room.get('submissions',{});rows=[];winners=[]
    if mode=='go':
        valid=sorted([(v['ms'],p) for p,v in subs.items() if p not in false]);pts={}
        for i,(ms,p) in enumerate(valid):pts[p]=3 if i==0 else 2 if i==1 else 1
        for p in pids:
            if p in false:choice='성급함';ok=False
            elif p in subs:choice=f'{subs[p]["ms"]}ms';ok=True
            else:choice='미반응';ok=False
            if p in pts:score(room,p,pts[p]);winners.append(p)
            rows.append({'player':room['players'][p]['name'],'choice':choice,'correct':ok,'fastest':bool(valid and p==valid[0][1]),'points':pts.get(p,0)})
        title=(room['players'][valid[0][1]]['name']+f' · {valid[0][0]}ms') if valid else '유효 반응 없음'
    else:
        for p in pids:
            safe=p not in false and p not in subs
            if safe:score(room,p,2);winners.append(p)
            rows.append({'player':room['players'][p]['name'],'choice':'참음' if safe else '눌러버림','correct':safe,'points':2 if safe else 0})
        title='STOP 함정 · '+(f'{len(winners)}명 생존' if winners else '전원 낚였습니다')
    room['round_result']={'stage':'reaction_result','title':title,'answers':rows,'winner_ids':winners};gd['reaction_stage']='result';room['public']['event_title']='결과';room['public']['event_text']=title;room['interaction_deadline']=now()+3.5;room['deadline']=room['interaction_deadline']

def next_reaction(room):
    gd=room['game_data'];gd['reaction_index']+=1
    if gd['reaction_index']>=gd['reaction_total']:finish_game(room)
    else:start_reaction_round(room)

    elif len(maf)>=len(town):

    else:return False

    room['game_data']['forced_winners']=winners;room['game_data']['result_label']=label;finish_game(room);return True

    if role not in ('마피아','경찰','의사'):raise ValueError('현재 역할에는 밤 행동이 없습니다.')
    if pid in gd.setdefault('mafia_actions',{}):raise ValueError('이미 밤 행동을 선택했습니다.')

    if all(p in gd['mafia_actions'] for p in required):resolve_mafia_night(room)

def mafia_vote(room,pid,target):
    gd=room['game_data'];alive=mafia_alive(room)

    room['votes'][pid]=target

_V45_SETUP_GAME=setup_game
def setup_game(room,replay=False):
    gid=room.get('game_id')
    if gid=='reaction':return setup_reaction(room)
    if gid=='mafia':return setup_mafia(room)
    return _V45_SETUP_GAME(room,replay)

_V45_ENTER_PLAY=enter_play
def enter_play(room):
    gid=room.get('game_id')
    if gid=='reaction':room['phase']='play';room['phase_started']=now();return start_reaction_round(room)
    if gid=='mafia':room['phase']='play';room['phase_started']=now();return start_mafia_night(room)
    return _V45_ENTER_PLAY(room)

_V45_DO_ACTION=do_action
def do_action(room,pid,data):
    t=data.get('type')
    if t=='reaction_tap':
        if room.get('phase')!='play' or room.get('game_id')!='reaction' or room.get('paused'):raise ValueError('지금은 반응할 수 없습니다.')
        return reaction_tap(room,pid)
    if t=='mafia_night_action':
        if room.get('phase')!='play' or room.get('game_id')!='mafia' or room.get('paused'):raise ValueError('지금은 밤 행동을 할 수 없습니다.')
        return mafia_night_action(room,pid,data.get('target'))
    if t=='mafia_vote':
        if room.get('phase')!='play' or room.get('game_id')!='mafia' or room.get('paused'):raise ValueError('지금은 투표할 수 없습니다.')
        return mafia_vote(room,pid,data.get('target'))
    return _V45_DO_ACTION(room,pid,data)

_V45_TICK=tick
def tick(room):
    _V45_TICK(room)
    if room.get('phase')!='play' or room.get('paused'):return
    gid=room.get('game_id');gd=room.get('game_data',{});ts=now()
    if gid=='reaction':
        st=gd.get('reaction_stage')
        if st=='wait' and ts>=gd.get('reaction_cue_at',1e18):cue_reaction(room)
        elif st=='cue' and ts>=gd.get('reaction_resolve_at',1e18):resolve_reaction(room)
        elif st=='result' and ts>=room.get('interaction_deadline',1e18):next_reaction(room)
    elif gid=='mafia':
        st=gd.get('mafia_stage')
        if st=='night' and ts>=room.get('interaction_deadline',1e18):resolve_mafia_night(room)
        elif st=='morning' and ts>=room.get('interaction_deadline',1e18):start_mafia_discuss(room)
        elif st=='discuss' and ts>=room.get('interaction_deadline',1e18):start_mafia_vote(room)
        elif st=='vote' and ts>=room.get('interaction_deadline',1e18):resolve_mafia_vote(room)
        elif st=='vote_result' and ts>=room.get('interaction_deadline',1e18):gd['mafia_day']+=1;start_mafia_night(room)
