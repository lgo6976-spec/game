def make_assignment(room,pid,role='플레이어',mission='',target=None,extra=None):
    prev=(room.get('assignments',{}).get(pid) or {}).get('extra',{})
    merged={'rerolls_left':int(prev.get('rerolls_left',1)),'assignment_version':uid()}
    merged.update(extra or {})
    merged['assignment_version']=uid()
    room['assignments'][pid]={'role':role,'mission':mission,'target':target,'extra':merged}

def setup_game(room, replay=False):

    if gid=='bounty':
        for i,pid in enumerate(pids): make_assignment(room,pid,'사냥꾼',target_mission(room),pids[(i+1)%len(pids)])

    ass=room.get('assignments',{}).get(pid)
    if not ass:return
    extra=ass.setdefault('extra',{});old=str(extra.get('private_notice','')).strip();lines=([x for x in old.splitlines() if x.strip()]+[str(text).strip()])[-3:];extra['private_notice']='\n'.join(lines)

def eligible_mission_pid(room,pid):
    ass=room.get('assignments',{}).get(pid) or {};role=ass.get('role');gid=room.get('game_id')
    if gid=='bounty':return role=='사냥꾼' and room['players'].get(pid,{}).get('alive') and ass.get('target')
    if gid=='bomb':return role=='💣 폭탄 보유' and ass.get('target')
    if gid=='infection':return role=='☣️ 감염자' and ass.get('target')
    if gid=='forbidden':return role=='유도자' and ass.get('target')
    return gid in ('secret','night','betray')

def apply_event_effect(room,ev):
    gd=room.setdefault('game_data',{})
    if '점수가 2배' in ev:gd['score_multiplier_until']=now()+(180 if '3분' in ev else 120)
    if '성공 점수가 +1점' in ev or '보너스 +1점' in ev or '+1점 추가' in ev:
        if '90초' in ev:gd['success_bonus_until']=now()+90
        else:gd['next_success_bonus']=max(1,int(gd.get('next_success_bonus',0)))
    if '새로운 비밀 지령' in ev:
        candidates=[p for p in active_pids(room) if eligible_mission_pid(room,p) and not any(x.get('from')==p for x in room.get('pending',{}).values())]
        if candidates:
            p=random.choice(candidates);ass=room['assignments'][p];ass['mission']=self_mission(room) if room['game_id'] in ('secret','night') else target_mission(room);ass.setdefault('extra',{})['assignment_version']=uid();append_private_notice(room,p,'시스템 이벤트로 새로운 지령이 도착했습니다.')
    if '블라인드 힌트' in ev:
        candidates=active_pids(room)
        if candidates:append_private_notice(room,random.choice(candidates),'블라인드 경고: 지금부터 60초 동안 평소보다 주변 행동을 더 의심하세요.')
    if '거짓말 카드' in ev:
        candidates=active_pids(room)
        if candidates:append_private_notice(room,random.choice(candidates),'거짓말 카드: 다음 질문 한 번에는 거짓말해도 됩니다.')
    if '단서 공개' in ev:
        story=room.get('public',{}).get('story') or {};candidates=active_pids(room);clues=story.get('clues') or []
        if candidates:append_private_notice(room,random.choice(candidates),'비밀 단서: '+(clues[0] if clues else story.get('twist') or '누군가의 행동 순서를 다시 확인하세요.'))
    if '빠른 라운드' in ev:gd['fast_verify_until']=now()+60

def trigger_event(room):
    if room.get('phase')!='play' or room.get('paused'):return
    gid=room['game_id'];is_clue=False
    if gid=='night':
        story=room.get('public',{}).get('story') or {};clues=story.get('clues') or [];idx=int(room.get('game_data',{}).get('story_clue_idx',0))
        if idx<len(clues) and random.random()<.34:
            ev=clues[idx];is_clue=True;room['game_data']['story_clue_idx']=idx+1;room['public']['event_title']=f'단서 {idx+1}';room['public']['event_text']=ev
        else:ev=random.choice(NIGHT_EVENTS);room['public']['event_title']='새로운 사건';room['public']['event_text']=ev
    else:ev=random.choice(GENERAL_EVENTS);room['public']['event_title']='새로운 사건';room['public']['event_text']=ev
    if not is_clue:apply_event_effect(room,ev)
    if '60초 감소' in ev and room.get('deadline'):room['deadline']=max(now()+5,room['deadline']-60)
    tempo=room.get('settings',{}).get('tempo','brisk')
    span=((85,115) if tempo=='fast' else (100,135) if tempo=='brisk' else (120,155)) if gid=='night' else ((55,85) if tempo=='fast' else (70,110) if tempo=='brisk' else (95,145));room['next_auto_event']=now()+random.randint(*span)

def finish_game(room):
    if room.get('phase')=='result':return
    gid=room['game_id'];result_label=room.get('game_data',{}).get('result_label');forced_winners=list(room.get('game_data',{}).get('forced_winners') or [])
    if gid=='bomb':
        holder=room['game_data'].get('holder')
        if holder and holder in room['players'] and room['players'][holder].get('active'):score(room,holder,-3);add_metric(room,holder,'bomb_end',1)
    if gid=='infection':
        infected=set(room['game_data'].get('infected',[]));humans=[p for p in active_pids(room) if p not in infected]
        if humans:
            for p in humans:score(room,p,3)
            result_label=f'인간팀 승리 · {len(humans)}명 생존';forced_winners=humans
        else:
            initial=[p for p in room['game_data'].get('initial_infected',[]) if p in room['players']]
            for p in initial:score(room,p,3)
            result_label='감염팀 승리 · 전원 감염';forced_winners=initial or active_pids(room)
    if gid=='night' and not room.get('round_result'):room['public']['event_title']='최종 투표';room['public']['event_text']='최종 투표가 완료되지 않아 이번 판 점수로 종료합니다.'
    starts=room.get('game_data',{}).get('start_scores',{});game_points={p:room['players'][p]['score']-starts.get(p,room['players'][p]['score']) for p in active_pids(room)}
    if forced_winners:winners=[p for p in forced_winners if p in game_points]
    else:
        maxscore=max(list(game_points.values()) or [0]);winners=[p for p,v in game_points.items() if v==maxscore]
    if not result_label:result_label=room['players'][winners[0]]['name'] if len(winners)==1 else (' · '.join(room['players'][p]['name'] for p in winners)+' 공동 1위' if winners else '이번 판 종료')
    for p in winners:add_metric(room,p,'wins',1)
    primary=winners[0] if winners else None;room['history'].append({'game':gid,'winner':primary,'winners':winners,'at':now(),'game_points':game_points});room['phase']='result';room['deadline']=None;room['interaction_deadline']=None;room['next_auto_event']=None;room['paused']=False;room['paused_at']=None;room['pending']={};room['result']={'winner':primary,'winner_ids':winners,'winner_label':result_label,'title':GAMES[gid]['title'],'game_points':game_points}

def ensure_host(room):
    players=room.get('players',{});active=[p for p in players.values() if p.get('active')]

    host=players.get(room.get('host_pid'));ts=now();recent=[p for p in active if ts-p.get('last_seen',0)<=15];stale=(not host or not host.get('active') or ts-host.get('last_seen',0)>45)
    if not stale or not recent:return False
    new_host=max(recent,key=lambda p:p.get('last_seen',0))
    if host and host.get('id')==new_host.get('id'):return False
    for p in players.values():p['host']=False
    new_host['host']=True;room['host_pid']=new_host['id'];room.setdefault('public',{})['event_title']='방장 자동 승계';room['public']['event_text']=f'{new_host["name"]}님이 새 방장이 되었습니다.';return True

def public_game_data(room,pid):
    gid=room['game_id'];gd=room['game_data'];out={}
    if gid=='infection':out['infection_rate']=round(100*len([p for p in gd.get('infected',[]) if p in active_pids(room)])/max(1,len(active_pids(room))))
    if gid=='bomb':out['bomb_moves']=sum(p['metrics'].get('bomb_pass',0) for p in room['players'].values())
    if gid=='night' and room['phase']=='play':out['votes_received']=Counter(v for voter,v in room['votes'].items() if voter in active_pids(room)).get(pid,0)
    if gid in ('oddq','anon','choice'):out['submitted']=len([p for p in room['submissions'] if p in active_pids(room)]);out['needed']=len(active_pids(room));out['my_submitted']=pid in room.get('submissions',{})
    if gid=='oddq' and (room.get('round_result') or {}).get('stage')=='guess':out['guessed']=len([p for p in room['votes'] if p in active_pids(room)]);out['needed']=len(active_pids(room));out['my_guessed']=pid in room.get('votes',{})
    out['interaction_total']=gd.get('interaction_total',0);return out

def do_action(room,pid,data):
    t=data.get('type');me=room['players'].get(pid)
    if not me:raise ValueError('참가자를 찾을 수 없습니다.')
    is_host=(pid==room['host_pid'])
    if t=='leave_room':
        if room.get('phase') in ('countdown','play'):raise ValueError('진행 중인 판에서는 나갈 수 없습니다. 게임 종료 후 나가 주세요.')
        me['active']=False;me['last_seen']=now()
        if is_host:
            others=[x for x in room['players'].values() if x['id']!=pid and x.get('active')]
            if others:me['host']=False;new_host=max(others,key=lambda x:x.get('last_seen',0));new_host['host']=True;room['host_pid']=new_host['id'];room['public']['event_title']='방장 변경';room['public']['event_text']=f'{new_host["name"]}님이 새 방장이 되었습니다.'
        return
    if t=='settings':
        if not is_host:raise ValueError('방장만 설정할 수 있습니다.')
        if room['phase'] in ('countdown','play'):raise ValueError('게임 중에는 설정을 바꿀 수 없습니다.')
        if 'spice' in data and data['spice'] in ('mild','spicy','hot'):room['settings']['spice']=data['spice']
        if 'tempo' in data and data['tempo'] in ('fast','brisk','relaxed'):room['settings']['tempo']=data['tempo']
        for k in ('touch','drink','dating','verification'):
            if k in data:room['settings'][k]=bool(data[k])
    elif t=='select_game':
        if not is_host:raise ValueError('방장만 게임을 선택할 수 있습니다.')
        if room['phase'] not in ('lobby','games','result','guide'):raise ValueError('진행 중인 게임에서는 게임을 바꿀 수 없습니다.')
        gid=data.get('game_id')
        if gid not in GAMES:raise ValueError('게임이 없습니다.')
        room['game_id']=gid;room['phase']='guide';room['round_result']=None;room['result']=None
    elif t=='open_games':
        if not is_host:raise ValueError('방장만 게임 선택 화면을 열 수 있습니다.')
        if room['phase'] in ('countdown','play'):raise ValueError('진행 중인 게임을 먼저 종료해 주세요.')
        room['phase']='games';room['deadline']=None
    elif t=='back_lobby':
        if not is_host:raise ValueError('방장만 이동할 수 있습니다.')
        if room['phase'] in ('countdown','play'):raise ValueError('진행 중인 게임을 먼저 종료해 주세요.')
        room['phase']='lobby';room['game_id']=None;room['deadline']=None
    elif t=='start':
        if not is_host:raise ValueError('방장만 시작할 수 있습니다.')
        if room['phase']!='guide' or not room.get('game_id'):raise ValueError('게임 설명 화면에서 시작해 주세요.')
        n=len(active_pids(room));lo,hi=game_player_limits(room['game_id'])
        if not(lo<=n<=hi):raise ValueError(f'{GAMES[room["game_id"]]["title"]}은(는) {lo}~{hi}명으로 플레이해 주세요. 현재 {n}명입니다.')
        setup_game(room)
    elif t=='replay':
        if not is_host:raise ValueError('방장만 재경기할 수 있습니다.')
        if room['phase']!='result':raise ValueError('결과 화면에서 재경기해 주세요.')
        n=len(active_pids(room));lo,hi=game_player_limits(room['game_id'])
        if not(lo<=n<=hi):raise ValueError(f'{GAMES[room["game_id"]]["title"]}은(는) {lo}~{hi}명으로 플레이해 주세요. 현재 {n}명입니다.')
        setup_game(room,True)
    elif t=='pause':
        if not is_host:raise ValueError('방장만 일시정지할 수 있습니다.')
        if room['phase']!='play':return
        if not room['paused']:
            ts=now();room['pause_remaining']=max(0,room['deadline']-ts);room['paused']=True;room['paused_at']=ts
        else:
            ts=now();paused_at=room.get('paused_at') or ts;delta=max(0,ts-paused_at);room['deadline']=ts+(room.get('pause_remaining') or 0)
            for key in ('interaction_deadline','next_auto_event'):
                if room.get(key):room[key]+=delta
            for p in room.get('pending',{}).values():
                if p.get('deadline'):p['deadline']+=delta
            for key in ('score_multiplier_until','success_bonus_until','fast_verify_until'):
                if room.get('game_data',{}).get(key):room['game_data'][key]+=delta
            for ass in room.get('assignments',{}).values():
                ra=(ass.get('extra') or {}).get('reveal_after')
                if ra:ass['extra']['reveal_after']=ra+delta
            room['pause_remaining']=None;room['paused']=False;room['paused_at']=None
    elif t=='event':
        if not is_host:raise ValueError('방장만 사건을 발생시킬 수 있습니다.')
        if room['phase']!='play' or room.get('paused'):raise ValueError('게임 진행 중에만 사건을 발생시킬 수 있습니다.')
        trigger_event(room)
    elif t=='end':
        if not is_host:raise ValueError('방장만 종료할 수 있습니다.')

def handle_submit(room,pid,data):
    gid=room['game_id']
    if room.get('round_result'):raise ValueError('현재 제출 단계가 아닙니다.')
    if pid in room.get('submissions',{}):raise ValueError('이미 제출했습니다. 다른 참가자를 기다려 주세요.')
    if gid in ('oddq','anon'):
        target=data.get('target')
        if target not in active_pids(room) or target==pid:raise ValueError('다른 참가자를 선택하세요.')
        room['submissions'][pid]={'target':target,'reason':str(data.get('reason',''))[:80]}
        if len([p for p in room['submissions'] if p in active_pids(room)])>=len(active_pids(room)):resolve_odd_answers(room) if gid=='oddq' else resolve_anon(room)
    elif gid=='choice':
        choice=data.get('choice')
        if choice not in ('a','b'):raise ValueError('선택이 필요합니다.')
        room['submissions'][pid]={'choice':choice}
        if len([p for p in room['submissions'] if p in active_pids(room)])>=len(active_pids(room)):resolve_choice(room)
    else:raise ValueError('현재 게임에서는 제출할 수 없습니다.')
