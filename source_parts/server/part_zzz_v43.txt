def ensure_v43(room):
    s=room.setdefault('settings',{})
    s.setdefault('small_rounds',8)
    s.setdefault('small_seconds',10)
    return room

def small_round_count(room): return max(5,min(12,int(room.get('settings',{}).get('small_rounds',8))))
def small_seconds(room): return max(6,min(20,int(room.get('settings',{}).get('small_seconds',10))))

def resolve_small(room):
    gid=room['game_id'];gd=room['game_data'];pids=active_pids(room);subs=room.get('submissions',{});rows=[];winners=[];title='라운드 종료'
    if gid=='match':
        vals={p:(subs.get(p) or {}).get('choice') for p in pids};cnt=Counter(v for v in vals.values() if v);mx=max(cnt.values() or [0]);unanimous=(mx==len(pids) and len(pids)>1)
        for p,v in vals.items():
            win=bool(v and ((len(pids)==2 and unanimous) or (len(pids)>=3 and cnt[v]==mx and mx>len(pids)/2)));pts=2 if unanimous and win else 1 if win else 0
            if pts: score(room,p,pts);winners.append(p)
            rows.append({'player':room['players'][p]['name'],'choice':v or '미응답','correct':win})
        title='완벽한 싱크!' if unanimous else ('다수파 성공' if winners else '이번엔 엇갈렸어요')
    elif gid=='teamsync':
        vals={p:(subs.get(p) or {}).get('choice') for p in pids};teamstats={}
        for team in ('A','B'):
            members=[p for p in pids if gd['teams'][p]==team];cnt=Counter(vals[p] for p in members if vals[p]);best=max(cnt.values() or [0]);rate=(best/len(members) if members else 0);teamstats[team]={'rate':rate,'members':members,'best':best}
        best_rate=max(teamstats['A']['rate'],teamstats['B']['rate']);winning_teams=[t for t in ('A','B') if teamstats[t]['rate']==best_rate and best_rate>0]
        for p,v in vals.items():
            team=gd['teams'][p];win=team in winning_teams
            if win: score(room,p,2 if len(winning_teams)==1 else 1);winners.append(p)
            rows.append({'player':room['players'][p]['name']+f' · 팀 {team}','choice':v or '미응답','correct':win})
        title=('팀 '+winning_teams[0]+' 승리' if len(winning_teams)==1 else '팀 싱크 동률')+f' · A {round(teamstats["A"]["rate"]*100)}% / B {round(teamstats["B"]["rate"]*100)}%'
    elif gid=='number':
        vals={p:int((subs.get(p) or {}).get('choice',0) or 0) for p in pids};valid={p:v for p,v in vals.items() if 1<=v<=9};rule=gd['number_rule'];winset=[]
        if valid:
            cnt=Counter(valid.values())
            if rule=='low_unique':
                u=[v for v,c in cnt.items() if c==1];target=min(u) if u else None;winset=[p for p,v in valid.items() if v==target]
            elif rule=='high_unique':
                u=[v for v,c in cnt.items() if c==1];target=max(u) if u else None;winset=[p for p,v in valid.items() if v==target]
            elif rule=='avg80':
                target=sum(valid.values())/len(valid)*.8;d=min(abs(v-target) for v in valid.values());winset=[p for p,v in valid.items() if abs(v-target)==d]
            elif rule=='median':
                arr=sorted(valid.values());m=len(arr);target=arr[m//2] if m%2 else (arr[m//2-1]+arr[m//2])/2;d=min(abs(v-target) for v in valid.values());winset=[p for p,v in valid.items() if abs(v-target)==d]
            elif rule=='pair': winset=[p for p,v in valid.items() if cnt[v]==2]
            elif rule=='solo': winset=[p for p,v in valid.items() if cnt[v]==1]
            elif rule=='closest5':
                d=min(abs(v-5) for v in valid.values());winset=[p for p,v in valid.items() if abs(v-5)==d]
            elif rule=='far5':
                d=max(abs(v-5) for v in valid.values());winset=[p for p,v in valid.items() if abs(v-5)==d]
        for p,v in vals.items():
            win=p in winset
            if win: score(room,p,2);winners.append(p)
            rows.append({'player':room['players'][p]['name'],'choice':str(v) if v else '미응답','correct':win})
        gd['last_number_values']=[valid[p] for p in winset if p in valid];title=(' · '.join(room['players'][p]['name'] for p in winset)+' 승리') if winset else '승자 없음'
    elif gid=='bluff':
        claim_id=gd.get('claim_id');claim=next((c for c in BLUFF_CLAIMS if c[0]==claim_id),None);truth=bool(claim and claim[2](gd['secret']));bluffer=gd['bluffer'];fooled=0
        for p in pids:
            if p==bluffer: continue
            ch=(subs.get(p) or {}).get('choice');guess=(ch=='truth');ok=(ch in ('truth','lie') and guess==truth)
            if ok: score(room,p,2);winners.append(p)
            elif ch in ('truth','lie'): fooled+=1
            rows.append({'player':room['players'][p]['name'],'choice':'진실' if ch=='truth' else '거짓' if ch=='lie' else '미응답','correct':ok})
        if fooled: score(room,bluffer,fooled);winners.append(bluffer)
        rows.insert(0,{'player':room['players'][bluffer]['name']+' · 블러퍼','choice':f'실제 숫자 {gd["secret"]}','correct':fooled>0});title=('진실이었다' if truth else '거짓이었다')+f' · {fooled}명 속음'
    elif gid=='chosung':
        q=gd['chosung']
        def norm(x): return ''.join(str(x).strip().lower().split())
        answers={norm(x) for x in q['answers']};correct=[]
        for p,v in subs.items():
            text=str(v.get('text',''));ok=norm(text) in answers
            if ok: correct.append((float(v.get('ts',now())),p,bool(v.get('after_hint'))))
        correct.sort()
        for rank,(_,p,after_hint) in enumerate(correct):
            pts=max(1,(3 if rank==0 else 2)-(1 if after_hint else 0));score(room,p,pts);winners.append(p)
        for p in pids:
            v=subs.get(p);text=(v or {}).get('text') or '미응답';ok=any(p==x[1] for x in correct);rows.append({'player':room['players'][p]['name'],'choice':text,'correct':ok,'fastest':bool(correct and p==correct[0][1])})
        title='정답 '+q['answers'][0]
    room['round_result']={'stage':'small_result','title':title,'answers':rows,'winner_ids':winners};gd['small_stage']='result';gd['interaction_total']=4;room['interaction_deadline']=now()+4;room['deadline']=room['interaction_deadline'];room['public']['event_title']='결과 공개';room['public']['event_text']=title

def next_small(room):
    gd=room['game_data'];gd['small_index']+=1
    if gd['small_index']>=gd['small_total']: finish_game(room)
    else: start_small_round(room)

_V43_TICK=tick
def tick(room):
    if room.get('game_id') not in SMALL_GAMES: return _V43_TICK(room)
    gd=room.get('game_data',{})
    if room.get('phase')=='countdown' and room.get('deadline') and now()>=room['deadline']:
        room['phase']='play';room['phase_started']=now();start_small_round(room);return
    if room.get('phase')!='play' or room.get('paused'): return
    if room['game_id']=='chosung' and gd.get('small_stage')=='answer' and not gd.get('hint_revealed') and now()>=gd.get('hint_at',1e18):
        gd['hint_revealed']=True;q=gd['chosung'];room['public']['event_text']='힌트: '+q['hint']
        for p in active_pids(room):
            a=room['assignments'].get(p)
            if a: a.setdefault('extra',{})['hint']=q['hint']
    if room.get('interaction_deadline') and now()>=room['interaction_deadline']:
        if gd.get('small_stage') in ('answer','vote'): resolve_small(room)
        elif gd.get('small_stage')=='claim':
            cid=random.choice(gd['claim_choices']);gd['claim_id']=cid;gd['small_stage']='vote';text=next(c[1] for c in BLUFF_CLAIMS if c[0]==cid);room['public']['event_text']=room['players'][gd['bluffer']]['name']+'의 주장: '+text;room['submissions']={};room['interaction_deadline']=now()+small_seconds(room);room['deadline']=room['interaction_deadline']
            for p in active_pids(room):
                if p!=gd['bluffer']: make_assignment(room,p,'판정자',text,None,{'vote_options':['truth','lie']})
        elif gd.get('small_stage')=='result': next_small(room)

_V43_HANDLE_SUBMIT=handle_submit
def handle_submit(room,pid,data):
    gid=room.get('game_id')
    if gid not in SMALL_GAMES: return _V43_HANDLE_SUBMIT(room,pid,data)
    gd=room['game_data'];stage=gd.get('small_stage')
    if gid=='bluff' and stage=='claim':
        if pid!=gd.get('bluffer'): raise ValueError('블러퍼만 주장을 선택할 수 있습니다.')
        cid=str(data.get('choice',''))
        if cid not in gd.get('claim_choices',[]): raise ValueError('제시된 주장 중 하나를 골라 주세요.')
        gd['claim_id']=cid;gd['small_stage']='vote';room['submissions']={};text=next(c[1] for c in BLUFF_CLAIMS if c[0]==cid);room['public']['event_text']=room['players'][pid]['name']+'의 주장: '+text
        for p in active_pids(room):
            if p!=pid: make_assignment(room,p,'판정자',text,None,{'vote_options':['truth','lie']})
        room['interaction_deadline']=now()+small_seconds(room);room['deadline']=room['interaction_deadline'];return
    if stage not in ('answer','vote'): raise ValueError('지금은 제출할 수 없습니다.')
    if pid in room.get('submissions',{}): raise ValueError('이미 선택했습니다.')
    if gid in ('match','teamsync'):
        choice=str(data.get('choice',''));opts=(room['assignments'].get(pid) or {}).get('extra',{}).get('options',[])
        if choice not in opts: raise ValueError('보기 중 하나를 선택해 주세요.')
        room['submissions'][pid]={'choice':choice,'ts':now()}
    elif gid=='number':
        try: v=int(data.get('choice'))
        except: raise ValueError('1~9 중 숫자를 골라 주세요.')
        if not 1<=v<=9 or v==gd.get('number_forbid'): raise ValueError('선택할 수 없는 숫자입니다.')
        room['submissions'][pid]={'choice':v,'ts':now()}
    elif gid=='bluff':
        if pid==gd.get('bluffer'): raise ValueError('블러퍼는 판정에 참여하지 않습니다.')
        ch=str(data.get('choice',''))
        if ch not in ('truth','lie'): raise ValueError('진실 또는 거짓을 선택해 주세요.')
        room['submissions'][pid]={'choice':ch,'ts':now()}
    elif gid=='chosung':
        text=str(data.get('text','')).strip()[:24]
        if not text: raise ValueError('정답을 입력해 주세요.')
        room['submissions'][pid]={'text':text,'ts':now(),'after_hint':bool(gd.get('hint_revealed'))}
    needed=len(active_pids(room))-(1 if gid=='bluff' else 0)
    if len(room['submissions'])>=needed: resolve_small(room)
