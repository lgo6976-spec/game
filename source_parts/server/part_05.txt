            expired.append((pend_id,p))
    for pend_id,p in expired:
        # Friendly party-game default: unanswered verification auto-approves to keep pace.
        apply_success(room,p['from'],p['target'],p['kind'])
        room['pending'].pop(pend_id,None)
        room['public']['event_title']='자동 확인'
        room['public']['event_text']='확인이 늦어 성공 신고가 자동 승인되었습니다.'

def tick(room):
    if room['phase']=='countdown' and room.get('deadline') and now()>=room['deadline']: enter_play(room)
    if room['phase']=='play' and not room.get('paused'):
        expire_pending(room)
        if room.get('interaction_deadline') and now()>=room['interaction_deadline']:
            auto_advance_interaction(room)
        if room.get('next_auto_event') and now()>=room['next_auto_event'] and room['game_id'] not in ('oddq','anon','choice'):
            trigger_event(room)
        # 펜션의 밤은 종료 3분 전 자동으로 최종 국면을 알려준다.
        if room['game_id']=='night' and room.get('deadline'):
            rem=room['deadline']-now()
            if rem<=180 and not room['game_data'].get('final_phase'):
                room['game_data']['final_phase']=True
                room['public']['event_title']='FINAL'
                room['public']['event_text']='3분 남았습니다. 이제 범인을 정하고 최종 투표를 완료하세요.'
        if room.get('deadline') and now()>=room['deadline']:
            if room['game_id']=='night' and room.get('votes'):
                resolve_night_vote(room)
            else:
                finish_game(room)


def cleanup_rooms():
    cutoff=now()-ROOM_TTL
    dead=[code for code,r in ROOMS.items() if r.get('last_access',r.get('created',0)) < cutoff]
    for code in dead: ROOMS.pop(code,None)
    return len(dead)

def save_rooms():
    try:
        STATE_FILE.parent.mkdir(parents=True,exist_ok=True)
        tmp=STATE_FILE.with_suffix(STATE_FILE.suffix+'.tmp')
        tmp.write_text(json.dumps(ROOMS,ensure_ascii=False,separators=(',',':')),encoding='utf-8')
        tmp.replace(STATE_FILE)
    except Exception:
        pass

def load_rooms():
    if not STATE_FILE.exists(): return
    try:
        data=json.loads(STATE_FILE.read_text(encoding='utf-8'))
        if isinstance(data,dict):
            ROOMS.update(data)
            cleanup_rooms()
    except Exception:
        pass

def public_origin(headers):
    proto=(headers.get('X-Forwarded-Proto') or 'http').split(',')[0].strip()
    host=(headers.get('X-Forwarded-Host') or headers.get('Host') or f'localhost:{PORT}').split(',')[0].strip()
    return f'{proto}://{host}'
