def active_pids(room): return [pid for pid,p in room['players'].items() if p.get('active',True)]
def filtered(items, settings):
    banned=set()
    if settings.get('touch'): banned.add('touch')
    if settings.get('drink'): banned.add('drink')
    if settings.get('dating'): banned.add('dating')
    return [x for x in items if not (set(x[-1]) & banned)]
def pick_recent(room,key,seq,identity=lambda x:x):
    if not seq: return None
    used=room['used'].setdefault(key,[])
    candidates=[x for x in seq if str(identity(x)) not in used]
    if not candidates:
        keep=max(1,len(used)//3)
        room['used'][key]=used[-keep:]
        used=room['used'][key]
        candidates=[x for x in seq if str(identity(x)) not in used] or list(seq)
    x=random.choice(candidates)
    used.append(str(identity(x)))
    max_keep=min(80,max(10,len(seq)*2))
    if len(used)>max_keep: room['used'][key]=used[-max_keep:]
    return x

def target_mission(room, kind=None, gid=None):
    """게임 속도에 맞는 전용 미션 풀에서 최근 중복을 피해서 하나를 고른다."""
    gid = gid or room.get('game_id') or 'bounty'
    pools = {
        'bounty': (BOUNTY_PHRASES, BOUNTY_ACTIONS, .45),
        'bomb': (BOMB_PHRASES, BOMB_ACTIONS, .58),
        'infection': (INFECTION_PHRASES, INFECTION_ACTIONS, .38),
        'forbidden': (FORBIDDEN_PHRASES, FORBIDDEN_ACTIONS, .48),
    }
    phrases, actions, say_ratio = pools.get(gid, (BOUNTY_PHRASES, BOUNTY_ACTIONS, .45))
    if kind is None:
        kind = 'say' if random.random() < say_ratio else 'action'
    if kind == 'say':
        phrase = pick_recent(room, f'{gid}_phrases', phrases, lambda x:x)
        return f'상대가 “{phrase}”라고 말하게 하세요.'
    pool = filtered(actions, room['settings'])
    x = pick_recent(room, f'{gid}_actions', pool, lambda a:a[0])
    return x[0]

def self_mission(room, gid=None):
    gid = gid or room.get('game_id') or 'secret'
    pool = NIGHT_MISSIONS if gid == 'night' else SECRET_MISSIONS
    pool = filtered(pool, room['settings'])
    x = pick_recent(room, f'{gid}_self_missions', pool, lambda a:a[0])
    return x[0]

def add_metric(room,pid,key,n=1):
    p=room['players'].get(pid)
    if p: p['metrics'][key]=p['metrics'].get(key,0)+n

def score(room,pid,n):
    if pid in room['players']: room['players'][pid]['score'] += n

def shuffled_active(room):
    p=active_pids(room)[:]; random.shuffle(p); return p

def make_assignment(room,pid,role='플레이어',mission='',target=None,extra=None):
    merged={'rerolls_left':1}
    merged.update(extra or {})
    room['assignments'][pid]={'role':role,'mission':mission,'target':target,'extra':merged}

def select_story(room):
    s=pick_recent(room,'stories',STORIES,lambda x:x[0])
    return {'title':s[0],'hook':s[1],'twist':s[2],'clues':STORY_CLUES_V3.get(s[0],[])}
