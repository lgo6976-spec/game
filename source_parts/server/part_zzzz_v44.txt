def ensure_v44(room):
    gd=room.setdefault('game_data',{})
    gd.setdefault('v44_success_times',[])
    gd.setdefault('v44_last_success_at',0)
    return room

def _mission_complexity(text):
    t=str(text);score=1
    for k in ('두 명','두 사람','두 번','연속','3분','5초 이상','두 개','두 사람의','각각'):
        if k in t:score+=1
    if len(t)>34:score+=1
    return min(3,score)

def _desired_mission_level(room,gid):
    gd=room.setdefault('game_data',{});ts=now();started=gd.get('v44_started_at') or room.get('phase_started') or ts
    recent=[x for x in gd.get('v44_success_times',[]) if ts-x<240]
    idle=ts-(gd.get('v44_last_success_at') or started)
    if idle>240:return 1
    if len(recent)>=4:return 3
    return 2

def handle_infection_scan(room,pid,target):
    gd=room.get('game_data',{});infected=set(gd.get('infected',[]))
    if room.get('game_id')!='infection' or not gd.get('scan_open'):raise ValueError('지금은 감염 검사 시간이 아닙니다.')
    if pid in infected:raise ValueError('감염자는 검사 투표에 참여할 수 없습니다.')
    if pid in gd.setdefault('scan_votes',{}):raise ValueError('이미 검사 대상을 선택했습니다.')
    if target not in active_pids(room) or target==pid:raise ValueError('다른 참가자를 선택하세요.')
    gd['scan_votes'][pid]=target
    humans=[p for p in active_pids(room) if p not in infected]
    if len(gd['scan_votes'])>=len(humans):resolve_infection_scan(room)

def anon_pass(room,pid):
    if room.get('game_id')!='anon' or room.get('round_result'):raise ValueError('지금은 패스할 수 없습니다.')
    if pid in room.get('submissions',{}):raise ValueError('이미 제출했습니다.')
    room['submissions'][pid]={'pass':True}
    if len([p for p in room['submissions'] if p in active_pids(room)])>=len(active_pids(room)):resolve_anon(room)

def _resolve_team_battle(room):
    gd=room['game_data'];mode=gd['team_mode'];subs=room.get('submissions',{});pids=active_pids(room);teams=gd['teams'];payload=gd['team_payload'];stats={};rows=[]
    for t in ('A','B'):stats[t]={'members':[p for p in pids if teams[p]==t],'metric':None}
    winners=[]
    if mode=='sync':
        for t in ('A','B'):
            vals=[(subs.get(p) or {}).get('choice') for p in stats[t]['members']];cnt=Counter(v for v in vals if v);stats[t]['metric']=max(cnt.values() or [0])/max(1,len(stats[t]['members']))
        best=max(stats[t]['metric'] for t in ('A','B'));winteams=[t for t in ('A','B') if stats[t]['metric']==best]
    elif mode=='number':
        target=payload['target']
        for t in ('A','B'):
            vals=[int((subs.get(p) or {}).get('choice',0) or 0) for p in stats[t]['members'] if (subs.get(p) or {}).get('choice')];avg=sum(vals)/len(vals) if vals else 99;stats[t]['metric']=abs(avg-target);stats[t]['avg']=avg
        best=min(stats[t]['metric'] for t in ('A','B'));winteams=[t for t in ('A','B') if stats[t]['metric']==best]
    elif mode=='quiz':
        ans=payload['answer']
        for t in ('A','B'):
            m=stats[t]['members'];ok=sum(1 for p in m if (subs.get(p) or {}).get('choice')==ans);stats[t]['metric']=ok/max(1,len(m))
        best=max(stats[t]['metric'] for t in ('A','B'));winteams=[t for t in ('A','B') if stats[t]['metric']==best]
    else:
        norm=lambda x:''.join(str(x).strip().lower().split());answers={norm(x) for x in payload['answers']}
        for t in ('A','B'):
            times=[float(v.get('ts',1e18)) for p,v in subs.items() if teams.get(p)==t and norm(v.get('text','')) in answers];stats[t]['metric']=min(times) if times else 1e18
        best=min(stats[t]['metric'] for t in ('A','B'));winteams=[t for t in ('A','B') if stats[t]['metric']==best and best<1e18]
        if not winteams:winteams=[]
    reward=3 if gd.get('team_final') else 2
    for p in pids:
        t=teams[p];win=t in winteams
        if win:score(room,p,reward);winners.append(p)
        v=subs.get(p) or {};choice=v.get('text') if mode=='chosung' else v.get('choice');rows.append({'player':room['players'][p]['name']+f' · 팀 {t}','choice':choice if choice not in (None,'') else '미응답','correct':win})
    if len(winteams)==1:title=f'팀 {winteams[0]} 승리 · +{reward}점'
    elif len(winteams)==2:title='동률 · 두 팀 모두 보상'
    else:title='이번 종목 승자 없음'
    room['round_result']={'stage':'small_result','title':title,'answers':rows,'winner_ids':winners};gd['small_stage']='result';gd['interaction_total']=4;room['interaction_deadline']=now()+4;room['deadline']=room['interaction_deadline'];room['public']['event_title']='팀 결과';room['public']['event_text']=title

def handle_submit(room,pid,data):
